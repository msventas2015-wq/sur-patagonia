SQL — Tabla proyectos

Ejecutá este SQL en el editor de Supabase (SQL Editor) para crear la tabla de proyectos. 

-- Ejecutar en Supabase > SQL Editor

CREATE TABLE IF NOT EXISTS proyectos (
  id              uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  nombre          text NOT NULL,
  slug            text UNIQUE NOT NULL,
  tagline         text,
  descripcion     text,
  categoria       text DEFAULT 'loteo',
  estado          text DEFAULT 'borrador',
  destacado       boolean DEFAULT false,
  activo          boolean GENERATED ALWAYS AS (estado = 'activo') STORED,

  -- Ubicación
  provincia       text,
  ciudad          text,
  distancia_km    numeric,
  tipo_acceso     text,
  acceso_desc     text,
  latitud         numeric,
  longitud        numeric,

  -- Superficie
  hectareas_total numeric,
  cantidad_lotes  integer,
  sup_min_m2      numeric,
  sup_max_m2      numeric,
  zonificacion    text,
  expediente      text,

  -- Reglamento
  altura_max      numeric,
  fos             numeric,
  reglamento_obs  text,

  -- Servicios (array de strings)
  servicios       text[] DEFAULT '{}',
  fecha_entrega   text,
  cronograma      text,

  -- Precios comerciales
  precio_desde    numeric,
  precio_hasta    numeric,
  monto_reserva   numeric,
  comision_pct    numeric,
  condiciones_venta text,
  tipo_propiedad  text,
  financiacion    text,
  financiacion_banco text DEFAULT 'no',
  ajuste_precios  text DEFAULT 'ninguno',

  -- Media
  foto_portada    text,
  fotos           text[] DEFAULT '{}',
  foto_atmosfera  text,
  foto_fondo      text,
  opacidad_fondo  integer DEFAULT 7,
  video_url       text,
  masterplan_url  text,
  brochure_url    text,
  tema            text,   -- JSON: {"paleta":"clasico","titulo":"cormorant","cuerpo":"montserrat"}

  -- Desarrollador (privado)
  dev_empresa     text,
  dev_cuit        text,
  dev_contacto_nombre text,
  dev_contacto_cargo text,
  dev_whatsapp    text,
  dev_email       text,
  dev_domicilio   text,
  notas_internas  text,

  -- Contacto público (miniweb)
  contacto_nombre   text,
  contacto_whatsapp text,
  contacto_email    text,

  created_at      timestamptz DEFAULT now(),
  updated_at      timestamptz DEFAULT now()
);

-- Tabla de lotes individuales
CREATE TABLE IF NOT EXISTS lotes (
  id              uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  proyecto_id     uuid REFERENCES proyectos(id) ON DELETE CASCADE,
  numero          text NOT NULL,
  superficie_m2   numeric,
  precio_usd      numeric,
  estado          text DEFAULT 'disponible', -- disponible | reservado | vendido
  frente_agua     boolean DEFAULT false,
  notas           text,
  created_at      timestamptz DEFAULT now(),
  updated_at      timestamptz DEFAULT now()
);

-- Índices
CREATE INDEX IF NOT EXISTS idx_proyectos_slug ON proyectos(slug);
CREATE INDEX IF NOT EXISTS idx_lotes_proyecto ON lotes(proyecto_id);
CREATE INDEX IF NOT EXISTS idx_lotes_estado ON lotes(estado);

-- Row Level Security (opcional pero recomendado)
ALTER TABLE proyectos ENABLE ROW LEVEL SECURITY;
ALTER TABLE lotes ENABLE ROW LEVEL SECURITY;

-- Política: lectura pública para proyectos activos
CREATE POLICY "proyectos_publicos" ON proyectos
  FOR SELECT USING (estado = 'activo');

-- Política: lectura pública de lotes de proyectos activos
CREATE POLICY "lotes_publicos" ON lotes
  FOR SELECT USING (
    EXISTS (SELECT 1 FROM proyectos p WHERE p.id = proyecto_id AND p.estado = 'activo')
  );
